Module HostLibs

AUTHOR
Alexey Morozov
PURPOSE
Dynamic library tools for Unix platforms

Summary

constantsNilLibHandle= NIL
type declarationsLibHandle =ADDRESS
procedures LoadLibrary (CONST fileName: ARRAY OF CHAR; VAR lib: LibHandle): BOOLEAN
FreeLibrary (CONST lib: LibHandle): BOOLEAN
GetProcedure (CONST lib: LibHandle; CONST name: ARRAY OF CHAR; CONST procAddr: ADDRESS): BOOLEAN

HostLibs.NilLibHandle

invalid library handle

Syntax

CONST NilLibHandle*= NIL

HostLibs.LibHandle

dynamic library handle type

Syntax

TYPE LibHandle* =ADDRESS

HostLibs.LoadLibrary

Load a dynamic library

Return
TRUE in case of success

Syntax

PROCEDURE LoadLibrary* (CONST fileName: ARRAY OF CHAR; VAR lib: LibHandle): BOOLEAN

Parameters

fileName
library file name
lib
returned loaded library handle; NilLibHandle in case of an error

HostLibs.FreeLibrary

Free a previously loaded dynamic library

Return
TRUE in case of success

Syntax

PROCEDURE FreeLibrary* (CONST lib: LibHandle): BOOLEAN

Parameters

lib
library handle

HostLibs.GetProcedure

Get a procedure from a loaded dynamic library

Return
TRUE in case of success

Syntax

PROCEDURE GetProcedure* (CONST lib: LibHandle; CONST name: ARRAY OF CHAR; CONST procAddr: ADDRESS): BOOLEAN

Parameters

lib
library handle
name
name of the procedure
procAddr
address of the destination procedure pointer (e.g. ADDRESSOF(procedureVariable))
